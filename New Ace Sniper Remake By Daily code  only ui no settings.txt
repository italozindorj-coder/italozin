--==================================================
-- HELPERS
--==================================================

local function Corner(object, radius)
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(0, radius)
	c.Parent = object
	return c
end

local function Stroke(object)
	local s = Instance.new("UIStroke")
	s.Color = BORDER
	s.Thickness = 1
	s.Transparency = 0.15
	s.Parent = object
	return s
end

local function Label(parent, text, size, bold)
	local t = Instance.new("TextLabel")

	t.BackgroundTransparency = 1
	t.Text = text
	t.TextColor3 = TEXT
	t.TextSize = size
	t.Font = bold and Enum.Font.GothamBold or Enum.Font.Gotham

	t.TextXAlignment = Enum.TextXAlignment.Left
	t.TextYAlignment = Enum.TextYAlignment.Center

	t.Parent = parent

	return t
end

--==================================================
-- DRAGGING
--==================================================

local dragging = false
local dragStart
local startPos

Window.InputBegan:Connect(function(input)

	if input.UserInputType == Enum.UserInputType.MouseButton1
	or input.UserInputType == Enum.UserInputType.Touch then

		dragging = true
		dragStart = input.Position
		startPos = Window.Position

	end

end)

UIS.InputChanged:Connect(function(input)

	if not dragging then
		return
	end

	if input.UserInputType == Enum.UserInputType.MouseMovement
	or input.UserInputType == Enum.UserInputType.Touch then

		local delta = input.Position - dragStart

		Window.Position = UDim2.new(
			startPos.X.Scale,
			startPos.X.Offset + delta.X,

			startPos.Y.Scale,
			startPos.Y.Offset + delta.Y
		)

	end

end)

UIS.InputEnded:Connect(function(input)

	if input.UserInputType == Enum.UserInputType.MouseButton1
	or input.UserInputType == Enum.UserInputType.Touch then

		dragging = false

	end

end)

	Row.BackgroundColor3 = ROW
	Row.BorderSizePixel = 0

	Stroke(Row)

	-- Text

	local Txt = Label(
		Row,
		text,
		7.5,
		false
	)

	Txt.Position = UDim2.fromOffset(9,0)
	Txt.Size = UDim2.new(1,-55,1,0)

	-- Toggle button

	local Button = Instance.new("TextButton")

	Button.Size = UDim2.fromOffset(32,18)
	Button.Position = UDim2.new(1,-40,0.5,-9)

	Button.Text = ""
	Button.AutoButtonColor = false

	Button.Parent = Row

	Corner(Button,20)

	-- Knob

	local Knob = Instance.new("Frame")

	Knob.Size = UDim2.fromOffset(14,14)

	Knob.Parent = Button

	Corner(Knob,20)

	-- Update

	local function Update()

		if enabled then

			Button.BackgroundColor3 =
				Color3.fromRGB(245,245,245)

			Knob.BackgroundColor3 =
				Color3.fromRGB(5,5,5)

			Knob.Position =
				UDim2.fromOffset(16,2)

		else

			Button.BackgroundColor3 =
				Color3.fromRGB(38,38,43)

			Knob.BackgroundColor3 =
				Color3.fromRGB(235,235,235)

			Knob.Position =
				UDim2.fromOffset(2,2)

		end

	end

	Update()

	Button.MouseButton1Click:Connect(function()

		enabled = not enabled
		Update()

	end)

end

local function Select(name)

	MainTab.BackgroundColor3 =
		name == "MAIN"
		and Color3.fromRGB(27,27,31)
		or Color3.fromRGB(9,9,11)

	FarmTab.BackgroundColor3 =
		name == "FARM"
		and Color3.fromRGB(27,27,31)
		or Color3.fromRGB(9,9,11)

	SettingsTab.BackgroundColor3 =
		name == "SETTINGS"
		and Color3.fromRGB(27,27,31)
		or Color3.fromRGB(9,9,11)

	if name == "MAIN" then

		MainPage()

	elseif name == "FARM" then

		FarmPage()

	elseif name == "SETTINGS" then

		SettingsPage()

	end

end

MainTab.MouseButton1Click:Connect(function()
	Select("MAIN")
end)

FarmTab.MouseButton1Click:Connect(function()
	Select("FARM")
end)

SettingsTab.MouseButton1Click:Connect(function()

end)
