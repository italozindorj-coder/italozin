-- Draggable logic (reconstructed from InputBegan / InputChanged log trace)
local function makeDraggable(frame)
    local dragging      = false
    local dragStart     = nil
    local startPosition = nil

    frame.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging      = true
            dragStart     = input.Position
            startPosition = frame.Position

            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)

    frame.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(
                startPosition.X.Scale,
                startPosition.X.Offset + delta.X,
                startPosition.Y.Scale,
                startPosition.Y.Offset + delta.Y
            )
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(
                startPosition.X.Scale,
                startPosition.X.Offset + delta.X,
                startPosition.Y.Scale,
                startPosition.Y.Offset + delta.Y
            )
        end
    end)
end

-- Helper: get the local player's Humanoid safely
local function getHumanoid()
    local char = LocalPlayer.Character
    if not char then return nil end
    return char:FindFirstChildOfClass("Humanoid")
end

-- Helper: get the local player's HumanoidRootPart safely
local function getHRP(char)
    char = char or LocalPlayer.Character
    if not char then return nil end
    return char:FindFirstChild("HumanoidRootPart")
end

-- ----------------------------------------------------------------
-- Anti Die: applies god-mode to the provided Humanoid
-- ----------------------------------------------------------------
local function applyAntiDie(humanoid)
    if not humanoid then return end

    -- Infinite health
    humanoid.MaxHealth = math.huge
    humanoid.Health    = math.huge

    -- Disable death state
    humanoid:SetStateEnabled(Enum.HumanoidStateType.Dead, false)

    -- Re-fill health any time it changes
    local conn1 = humanoid:GetPropertyChangedSignal("Health"):Connect(function()
        if CONFIG.ToggleStates.AntiDie and humanoid and humanoid.Parent then
            humanoid.Health = humanoid.MaxHealth
        end
    end)

    -- Re-fill health on state transition (e.g. fling damage triggers)
    local conn2 = humanoid.StateChanged:Connect(function(_, newState)
        if CONFIG.ToggleStates.AntiDie and humanoid and humanoid.Parent then
            if newState == Enum.HumanoidStateType.Dead then
                humanoid:SetStateEnabled(Enum.HumanoidStateType.Dead, false)
                humanoid.Health = humanoid.MaxHealth
            end
        end
    end)

    table.insert(AntiDieConnections, conn1)
    table.insert(AntiDieConnections, conn2)
end

-- ----------------------------------------------------------------
-- Anti Die toggle handler
-- ----------------------------------------------------------------
local function setAntiDie(state)
    CONFIG.ToggleStates.AntiDie = state

    -- Update toggle visual
    local knobColor = state and CONFIG.Colors.ToggleOnKnob  or CONFIG.Colors.ToggleOffKnob
    local bgColor   = state and CONFIG.Colors.ToggleOnBG    or CONFIG.Colors.ToggleOffBG
    AntiDieKnob.BackgroundColor3    = knobColor
    AntiDieToggleBG.BackgroundColor3 = bgColor

    -- Disconnect old connections
    for _, c in ipairs(AntiDieConnections) do
        c:Disconnect()
    end
    AntiDieConnections = {}

    local char = LocalPlayer.Character

    if state then
        -- Add green highlight to character
        if char then
            local existingHighlight = char:FindFirstChild(CONFIG.Names.HighlightTag)
            if existingHighlight then existingHighlight:Destroy() end

            local highlight = Instance.new("Highlight")
            highlight.Name               = CONFIG.Names.HighlightTag
            highlight.FillColor          = CONFIG.Colors.HighlightFill
            highlight.OutlineColor       = CONFIG.Colors.HighlightOutline
            highlight.FillTransparency   = 0.75
            highlight.OutlineTransparency = 0
            highlight.DepthMode          = Enum.HighlightDepthMode.AlwaysOnTop
            highlight.Parent             = char
        end

        -- Apply to current humanoid
        applyAntiDie(getHumanoid())

        -- Re-apply on respawn
        local conn = LocalPlayer.CharacterAdded:Connect(function(newChar)
            task.wait(0.1)
            if not CONFIG.ToggleStates.AntiDie then return end

            -- Re-add highlight
            local existingHighlight = newChar:FindFirstChild(CONFIG.Names.HighlightTag)
            if existingHighlight then existingHighlight:Destroy() end

            local highlight = Instance.new("Highlight")
            highlight.Name               = CONFIG.Names.HighlightTag
            highlight.FillColor          = CONFIG.Colors.HighlightFill
            highlight.OutlineColor       = CONFIG.Colors.HighlightOutline
            highlight.FillTransparency   = 0.75
            highlight.OutlineTransparency = 0
            highlight.DepthMode          = Enum.HighlightDepthMode.AlwaysOnTop
            highlight.Parent             = newChar

            -- Re-apply god mode
            local newHumanoid = newChar:WaitForChild("Humanoid", 5)
            applyAntiDie(newHumanoid)
        end)

        table.insert(AntiDieConnections, conn)
    else
        -- Remove highlight and restore normal health
        if char then
            local h = char:FindFirstChild(CONFIG.Names.HighlightTag)
            if h then h:Destroy() end

            local humanoid = getHumanoid()
            if humanoid then
                humanoid.MaxHealth = 100
                humanoid.Health    = 100
            end
        end
    end
end

-- ----------------------------------------------------------------
-- Anti Fling: claim network ownership of all players' HRPs
-- ----------------------------------------------------------------
local function claimOwnership(character)
    if not character then return end
    local hrp = character:FindFirstChild("HumanoidRootPart")
    if hrp then
        pcall(function()
            hrp:SetNetworkOwner()
        end)
    end
end

local function setAntiFling(state)
    CONFIG.ToggleStates.AntiFling = state

    -- Update toggle visual
    local knobColor = state and CONFIG.Colors.ToggleOnKnob  or CONFIG.Colors.ToggleOffKnob
    local bgColor   = state and CONFIG.Colors.ToggleOnBG    or CONFIG.Colors.ToggleOffBG
    AntiFlingKnob.BackgroundColor3    = knobColor
    AntiFlingToggleBG.BackgroundColor3 = bgColor

    -- Disconnect old connections
    for _, c in ipairs(AntiFlingConnections) do
        c:Disconnect()
    end
    AntiFlingConnections = {}

    if state then
        -- Apply to all current players
        for _, player in ipairs(Players:GetPlayers()) do
            claimOwnership(player.Character)

            -- Re-apply on their respawn
            local conn = player.CharacterAdded:Connect(function(char)
                task.wait(0.1)
                if CONFIG.ToggleStates.AntiFling then
                    claimOwnership(char)
                end
            end)
            table.insert(AntiFlingConnections, conn)
        end

        -- Handle players who join after Anti Fling is enabled
        local addedConn = Players.PlayerAdded:Connect(function(player)
            if not CONFIG.ToggleStates.AntiFling then return end
            claimOwnership(player.Character)
            local conn = player.CharacterAdded:Connect(function(char)
                task.wait(0.1)
                if CONFIG.ToggleStates.AntiFling then
                    claimOwnership(char)
                end
            end)
            table.insert(AntiFlingConnections, conn)
        end)
        table.insert(AntiFlingConnections, addedConn)

        -- Continuously re-assert ownership on RenderStepped (prevent bypass)
        local renderConn = RunService.RenderStepped:Connect(function()
            if not CONFIG.ToggleStates.AntiFling then return end
            for _, player in ipairs(Players:GetPlayers()) do
                claimOwnership(player.Character)
            end
        end)
        table.insert(AntiFlingConnections, renderConn)
    end
end

    tween:Play()
end
