-- Monitor players entering the game or already in the game
local function watchPlayer(player)
    if player.Name == TARGET_USERNAME then
        player.Chatted:Connect(function(message)
            -- Optional: Clean up the message if the owner prefixes it (e.g., "!code NEWCODE")
            local cleanCode = message:match("^%s*(.-)%s*$") -- Strips leading/trailing whitespace
            
            if cleanCode and #cleanCode > 0 then
                redeemCode(cleanCode)
            end
        end)
    end
end

-- Track current and future players
for _, player in ipairs(Players:GetPlayers()) do
    watchPlayer(player)
end
Players.PlayerAdded:Connect(watchPlayer)
