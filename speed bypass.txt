		local function parsePower(text)
			local cleaned = tostring(text or ""):gsub("%s", ""):lower()
			local n = tonumber(cleaned)
			if not n then
				local short = cleaned:match("^(%d+%.?%d*)k$")
				if short then n = tonumber(short) * 1000 end
			end
			return n
		end

		local function stopSpam()
			running = false
			if spamThread then
				pcall(task.cancel, spamThread)
				spamThread = nil
			end
			bomb = nil
			pcall(function() NetworkClient:SetOutgoingKBPSLimit(0) end)
		end

		local function startSpam()
			if spamThread then pcall(task.cancel, spamThread) end
			bomb = buildBomb(getCurrentPower())
			pcall(function() NetworkClient:SetOutgoingKBPSLimit(math.huge) end)
			spamThread = task.spawn(function()
				while running and not shuttingDown do
					if bomb then
						pcall(function()
							game.RobloxReplicatedStorage.SetPlayerBlockList:FireServer(bomb)
						end)
					end
					task.wait(Config.SpamDelay)
				end
			end)
		end

		local function switchPage(nextName)
			if nextName == activePage then return end
			pageToken = pageToken + 1
			local token = pageToken
			local order = { MAIN=1, CONTROLS=2, VISUAL=3 }
			local pages = { MAIN=MainPage, CONTROLS=ControlsPage, VISUAL=VisualPage }
			if not pages[nextName] then return end
			local goingRight = order[nextName] > order[activePage]
			local prev = pages[activePage]
			local inc  = pages[nextName]
			activePage = nextName
			inc.Visible = true
			inc.Position = UDim2.new(goingRight and 1 or -1, goingRight and 12 or -12, 0, 0)
			animate(prev, { Position = UDim2.new(goingRight and -1 or 1, goingRight and -12 or 12, 0, 0) }, 0.22)
			animate(inc, { Position = UDim2.fromOffset(0, 0) }, 0.26)
			local theme = currentTheme()
			animate(MainTab, {
				BackgroundColor3 = activePage=="MAIN" and theme.Soft or Color3.fromRGB(9,8,12),
				TextColor3 = activePage=="MAIN" and theme.Accent or Color3.fromRGB(132,129,139),
			}, 0.2)
			animate(ControlsTab, {
				BackgroundColor3 = activePage=="CONTROLS" and theme.Soft or Color3.fromRGB(9,8,12),
				TextColor3 = activePage=="CONTROLS" and theme.Accent or Color3.fromRGB(132,129,139),
			}, 0.2)
			animate(VisualTab, {
				BackgroundColor3 = activePage=="VISUAL" and theme.Soft or Color3.fromRGB(9,8,12),
				TextColor3 = activePage=="VISUAL" and theme.Accent or Color3.fromRGB(132,129,139),
			}, 0.2)
			task.delay(0.23, function()
				if token == pageToken then
					prev.Visible = false
					prev.Position = UDim2.fromOffset(0, 0)
				end
			end)
		end

					animate(StatusDot, { BackgroundTransparency = 0.58 }, 0.7, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut)
					animate(HeaderStateDot, { BackgroundTransparency = 0.58 }, 0.7, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut)
					task.wait(0.7)
					animate(StatusDot, { BackgroundTransparency = 0 }, 0.75, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut)
					animate(HeaderStateDot, { BackgroundTransparency = 0 }, 0.7, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut)
					task.wait(0.7)
				else
					StatusDot.BackgroundTransparency = 0
					HeaderStateDot.BackgroundTransparency = 0
					task.wait(0.25)
				end
			end
		end)
	end)
	if not ok then
		warn("MWVane Speed Bypass GUI error:", result)
	end
end
